import os
from Bio.PDB.Polypeptide import *
from Bio.PDB import *
from Bio.SeqIO import write
from Bio.Seq import Seq
from Bio.SeqRecord import SeqRecord

# %% cell 3 [code]
# This function will take a biopython structure object, get the PDB Id from that object, set fasta name, and save that fasta to fasta_location
def write_sequences_fasta(structure,fasta_directory):
    seq = []
    model = structure[0]
    # Ideally there will be only one chain
    for chain in model:
        for residue in chain:
            seq.append(three_to_one(residue.get_resname()))
    seq = Seq("".join(seq))
    record = SeqRecord(seq,id=structure.get_id())
    write(record,os.path.join(fasta_directory,f"{structure.get_id()}.fasta"),"fasta")

# %% cell 4 [code]
pdbDirectory = "S_669_PDB_Files"
fasta_directory = "./"
parser = PDBParser(QUIET=True)

for filename in os.listdir(pdbDirectory):
    structure = parser.get_structure(id=filename.split(".")[0],file=os.path.join(pdbDirectory,filename))
    write_sequences_fasta(structure,fasta_directory)
