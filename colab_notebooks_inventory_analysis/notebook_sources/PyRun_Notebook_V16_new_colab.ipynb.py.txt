def write_sequences_fasta(structure_dict, fasta_location):
    # This type of local import so that the import area does not get unnecessarily cluttered
    from Bio.SeqIO import write
    from Bio.Seq import Seq
    from Bio.SeqRecord import SeqRecord

    records = []

    for key, val in structure_dict.items():
        seq = []
        model = val[0]
        for chain in model:
            for residue in chain:
                seq.append(threeToOne(residue.get_resname()))
        seq = Seq("".join(seq))
        record = SeqRecord(seq, id=key)
        records.append(record)
        print(key)
    write(records, fasta_location, "fasta")

# In rare cases, some PDB files number chains with 1,2,3 instead of A,B,C
def convertChainFromAlphabetToNumber(alphabet):
    mappingDict = {ch: (idx + 1) for idx, ch in enumerate(ascii_uppercase)}
    return str(mappingDict[alphabet])

# Receives two Biopython residue objects and calculates the distance between them
def calc_residue_dist(residue_one, residue_two, atom="CB"):
    try:
        distance = residue_one[atom] - residue_two[atom]
    except KeyError:
        distance = residue_one["CA"] - residue_two["CA"]
    return distance


# Receives one Biopython residue object ``center`` and a list of residue objects ``neighbors``
# Calculates the distances between ``center`` and every residue from ``neighbors``
def calc_distances_for_neighbor_list(center, neighbors):
    dist_list = []
    for neighbor in neighbors:
        dist_list.append(calc_residue_dist(residue_one=center, residue_two=neighbor))
    return dist_list

    pose_dict = {}

# This function will receive a dictionary of pyrosetta poses, a dictionary of biopython structure
# and a list of (PdbId+ChainId)s where disparity is occurring between poses and structures
# For this version (an extension of the version above), the resolution process is to detect one specific residue that is present in the structure, but not in the pose
# And remove that specific residue from the structure object
# This function works with the assumption that the only difference between structure and pose can be one insertion in the structure
# This function makes one additional assumption that, each of the poses contain a single chain and each of the structures also contain a single chain
def resolvePoseStructureDisparity(pose_dict, structure_dict, disparities):
    for filename in disparities:
        pose = pose_dict[filename]
        structure = structure_dict[filename]

        pose_seq = pose.sequence()
        pose_len = pose.total_residue()

        structure_seq = []

        counter = 0

        # Create a local biopython numbering (0-based numbering) to residue object single-level dictionoary
        biopython_number_to_residue_map = {}
        model = structure[0]
        for chain in model:
            for residue in chain:
                structure_seq.append(threeToOne(residue.get_resname()))
                biopython_number_to_residue_map[str(counter)] = residue
                counter = counter + 1

        structure_seq = "".join(structure_seq)
        structure_len = counter

        # At first check which position in the biopython structure is extra
        structure_extra_index = 0
        for idx, bio_res in enumerate(structure_seq):
            # For the case when bio structure has one extra residue in the end
            if idx > (pose_len - 1):
                structure_extra_index = idx
                break
            elif pose_seq[idx] != bio_res:
                structure_extra_index = idx
                break

        # Now, remove the residue at ```structure_extra_index``` from the biopython structure
        residue_to_remove = biopython_number_to_residue_map[str(structure_extra_index)]
        # Single chain assumption coming in here
        chain = list(model.get_chains())[0]
        # Directly removing residue from a structure object from the structure_dict passed onto this function
        chain.detach_child(residue_to_remove.get_id())

# This function will receive a dictionary containing biopython structure objects as vals, where keys are (PdbId+ChainId)

# Adding icode to key of inner dict; if icode is empty, "Z" will be used as icode
def createTwoLevelMap(structure_dict):
    two_level_dict = {}
    print("Start creating two-level dictionary")
    for pdbIdPlusChain, structure in tqdm(structure_dict.items()):
        inner_dict = {}
        model = structure[0]
        counter = 0
        # Doing this because of the single chain assumption
        chain = list(model.get_chains())[0]
        for residue in chain:
            seqres = str(residue.get_id()[1])
            name = threeToOne(residue.get_resname())
            icode = residue.get_id()[2] if residue.get_id()[2] != " " else "Z"
            icode = str(icode)
            inner_dict[name + seqres + icode] = counter
            counter = counter + 1
        two_level_dict[pdbIdPlusChain] = copy.deepcopy(inner_dict)

    print("Done creating two-level dictionary")
    return two_level_dict

# 2D numpy array containing pairwise distances
def calc_all_pair_distances(residue_list):
    n_reisdues = len(residue_list)
    pair_distances = np.zeros((n_reisdues,n_reisdues))
    for i, c in enumerate(residue_list):
        is_c_GLY = (c.get_resname()== "GLY")
        for j, r in enumerate(residue_list):
            is_r_GLY = (r.get_resname()=="GLY")
            if (not is_c_GLY) and (not is_r_GLY):
                try:
                    c_coord = c["CB"].coord
                    r_coord = r["CB"].coord
                    pair_distances[i,j] = np.linalg.norm(c_coord-r_coord)
                except:
                    c_coord = c["CA"].coord
                    r_coord = r["CA"].coord
                    pair_distances[i,j] = np.linalg.norm(c_coord-r_coord)
            else:
                c_coord = c["CA"].coord
                r_coord = r["CA"].coord
                pair_distances[i,j] = np.linalg.norm(c_coord-r_coord)
    return pair_distances

# Receives one Biopython residue object ``center`` and a list of residue objects ``neighbors``
# Needs the ``two_level_dict`` for calculating PyRosetta pose numbering from biopython residue
# Needs ``pose_dict`` for getting the pyrosetta pose object to be sent to RosettaJumpDecorator
# Calculates the jump features between ``center`` and every residue from ``neighbors``
def calc_jump_features_for_neighbor_list(two_level_dict, pose_dict, pdbIdPlusChain, center, neighbors, ret_center_icode = True):
    jump_feature_list = []
    jump_decorator = RosettaJumpDecorator()
    res_one_name = threeToOne(center.get_resname())
    res_one_resseq = center.get_id()[1]
    res_one_icode = center.get_id()[2] if center.get_id()[2] != " " else "Z"
    res_one_icode = str(res_one_icode)
    for neighbor in neighbors:
        # Only three of the six vectors can be tried also as jump features
        res_two_name = threeToOne(neighbor.get_resname())
        res_two_resseq = neighbor.get_id()[1]
        res_two_icode = neighbor.get_id()[2] if neighbor.get_id()[2] != " " else "Z"
        res_two_icode = str(res_two_icode)
        resid1 = (two_level_dict[pdbIdPlusChain])[res_one_name + str(res_one_resseq) + res_one_icode] + 1
        resid2 = (two_level_dict[pdbIdPlusChain])[res_two_name + str(res_two_resseq) + res_two_icode] + 1
        jump_feature_list.append(
            jump_decorator.calc_edge_features(pose=pose_dict[pdbIdPlusChain], resid1=resid1, resid2=resid2))
    # This ``jump_feature_list`` is a list of (12,) shaped numpy arrays
    # Each element of the list represents relation with one specific neighbor
    # The 12 numbers describe two (translation+rotation) transformations
    # First 6 numbers are for center->neighbor transformation
    # Last 6 numbers are for neighbor->center transformation
    if ret_center_icode:
        return jump_feature_list, res_one_icode
    else:
        return jump_feature_list

def calc_hbond_features_for_neighbor_list(two_level_dict, pose_dict, pdbIdPlusChain, center, neighbors):
    pose = pose_dict[pdbIdPlusChain]
    hbond_feature_list = []
    hbond_decorator = RosettaHBondDecorator(pose=pose)

    return hbond_feature_list

def get_edge_index_induced(d_mat,cutoff):
    n_residues = len(d_mat)
    edge_indices = []
    for i in range(len(d_mat)):
        for j in range(len(d_mat)):
            if i!=j:
                if d_mat[i,j] <= cutoff:
                    # Self loops are currently added
                    edge_indices.append([i,j])
    return np.array(edge_indices,dtype=np.int64).T

def get_edge_index_complete(n_nodes):
    edge_indices = np.zeros((2,n_nodes*(n_nodes-1)),dtype=np.int64)
    column = 0
    for i in range(n_nodes):
        for j in range(n_nodes):
            if i!=j: 
                edge_indices[0,column] = i
                edge_indices[1,column] = j
                column = column + 1
    return edge_indices

# Assuming that, edge_index has shape (2,n_edges)
def get_edge_features(edge_index,residue_list,two_level_dict,pose_dict,pdbIdPlusChain,d_mat):
    edge_features = []
    # Doing transpose here since edge_index is of shape(2,n_edges),and we need to extract the edges one by one here  
    for edge in edge_index.T:
        first_res = residue_list[edge[0]]
        second_res = residue_list[edge[1]]
        dist_feature = np.array([d_mat[edge[0],edge[1]]])
        jump_features = calc_jump_features_for_pair(two_level_dict,pose_dict,pdbIdPlusChain,first_res,second_res)
        hbond_features = calc_hbond_features_for_pair(two_level_dict,pose_dict,pdbIdPlusChain,first_res,second_res)
        edge_features.append(np.concatenate((dist_feature,jump_features,hbond_features)))
    return np.array(edge_features,dtype=np.float32)



# The ``structure_dict`` argument contains pdbIdPlusChain as key and biopython structure objects as vals
# The ``position`` argument is pdb resseq and the ``amino_acid`` argument is a one-letter amino acid code
# This function will return two numpy arrays
# First one of size (#neighbors,#props/embedding_dimension), and second one of size(#neighbors,#edge_features)
# The ``first_matrix`` can be thought of as feature matrix and the ``second_matrix`` can be thought of as edge matrix
# The ``jump_matrix`` has shape (#neighbors,#(3+n_rotation-parameters)*2) where the first set of numbers are for
# the transformation from center to neighbor and the second set of numbers are for the transformation from neighbor to center
#  The ``hbond_matrix`` has shape (#neighbors,#hbond_features(5 currently))
no_neighbors = False
count = 0
# The ```residue_missing``` flag is to indicate situations when some residue has been removed from the system for whatever reason, and that residue
# wildType mutations need to be discarded
# Sending the ```ddgun_dict``` for adding 20 ddgun features to each of the neighbors 
# After this, the neighbors will have 43 features each (4 from property, 20 from PSSM, 19 from DDGun) 
residue_missing = False
def getNeighborMatrices(two_level_dict, pose_dict, structure_dict, ddgun_dict, pdbIdPlusChain, position, icode, amino_acid, pssm_dir,
                        distance=8):
    center, neighbors = getNeighbors(structure_dict, pdbIdPlusChain, position, icode, amino_acid, distance)
    distances = calc_distances_for_neighbor_list(center, neighbors)
    global residue_missing
    if not isinstance(center,Bio.PDB.Residue.Residue):
        print("CENTER "+str(center))
        print("INFORMATION " + pdbIdPlusChain+"->"+str(position)+"->"+amino_acid)
        residue_missing = True
        return 0
    else:
        residue_missing = False
    # This "center_icode" will be used in the calling function
    jump_features, center_icode  = calc_jump_features_for_neighbor_list(two_level_dict, pose_dict, pdbIdPlusChain, center, neighbors, ret_center_icode=True)
    hbond_features = calc_hbond_features_for_neighbor_list(two_level_dict, pose_dict, pdbIdPlusChain, center, neighbors)

    first_matrix = []
    pssm_matrix = []
    second_matrix = []
    jump_matrix = []
    hbond_matrix = []

    #GNN edge information start
    pairwise_distances = calc_all_pair_distances(residue_list=neighbors)
    n_edge_index_induced = get_edge_index_induced(d_mat=pairwise_distances,cutoff=distance)
    n_edge_index_complete = get_edge_index_complete(n_nodes=len(neighbors))
    n_edge_features_induced = get_edge_features(edge_index=n_edge_index_induced,residue_list=neighbors,
        two_level_dict=two_level_dict,pose_dict=pose_dict,pdbIdPlusChain=pdbIdPlusChain,d_mat=pairwise_distances)
    n_edge_features_complete = get_edge_features(edge_index=n_edge_index_complete,residue_list=neighbors,
        two_level_dict=two_level_dict,pose_dict=pose_dict,pdbIdPlusChain=pdbIdPlusChain,d_mat=pairwise_distances)

    gnn_edge_dict = {}
    gnn_edge_dict["edge_index_induced"] = n_edge_index_induced
    gnn_edge_dict["edge_index_complete"] = n_edge_index_complete
    gnn_edge_dict["edge_features_induced"] = n_edge_features_induced
    gnn_edge_dict["edge_features_complete"] = n_edge_features_complete
    #GNN edge information end

    for neighbor, distance, jump_feature, hbond_feature in zip(neighbors, distances, jump_features, hbond_features):
        first_matrix.append(getProps(threeToOne(neighbor.get_resname())))
        second_matrix.append(distance)
        jump_matrix.append(jump_feature)
        hbond_matrix.append(hbond_feature)

    pssm_matrix = get_pssmfeatures_for_neighbor_list(two_level_dict, pdbIdPlusChain, neighbors, pssm_dir)

    # dtype float32 is important because pytorch uses 32 bit floats instead of 64 bit ones
    # reshaping the second_matrix because currently, it only has edge feature
    first_matrix, second_matrix, jump_matrix, hbond_matrix = np.array(first_matrix, dtype=np.float32), \
                                                             np.array(second_matrix, dtype=np.float32).reshape(-1,
                                                                                                               1), np.array(
        jump_matrix, dtype=np.float32), \
                                                             np.array(hbond_matrix, dtype=np.float32)
    pssm_matrix = np.array(pssm_matrix, dtype=np.float32)

    # Start extracting DDGun features here
    # All the values will be populated with 0 and skip will become True in case ddgun is not available for one of the neighbors or all neighbors
    ddgun_pred_protchain = ddgun_dict.get(pdbIdPlusChain,None)

    if ddgun_pred_protchain is not None:
        ddgun_matrix = np.array(get_ddgunfeatures_for_neighbor_list(ddgun_pred_protchain, neighbors, fill_zero = False), dtype=np.float32)
    else:
        ddgun_matrix = np.array(get_ddgunfeatures_for_neighbor_list(None, neighbors, fill_zero = True), dtype=np.float32)


    # Something has to be done for the case when there are 0 neighbors, atleast add 1 neighbor right here with all 0 features
    # Since 0 padding is done during batch padding, this should not create any problems
    # Using some static values here for the dimensions of the arrays based on the anticipated number of features
    # This has to be changed and maybe passed as a parameter when array dimensions are supposed to changed due to changing number of features
    # In node embedding or edge features
    global no_neighbors
    global count
    if len(neighbors) == 0:
        # print("PdbId : {}, Position : {}".format(pdbIdPlusChain, position))
        # # 1 for distance
        # second_matrix = np.zeros((1, 1))
        # # 12 for 12 jump features
        # jump_matrix = np.zeros((1, 12))
        # hbond_matrix = np.zeros((1, 5))
        # # 4 since we are currently considering 4 properties per amino acid, this can change rapidly anytime
        # first_matrix = np.zeros((1, 4))
        # # 20 values for the PSSM features
        # pssm_matrix = np.zeros((1, 20))
        no_neighbors = True
        count = count + 1
        # Skipping the no_neighbor cases for now
        return 0
    else:
        no_neighbors = False

    # second_matrix contains edge features, and first_matrix contains node features
    second_matrix = np.concatenate((second_matrix, jump_matrix, hbond_matrix), axis=1)
    first_matrix = np.concatenate((first_matrix, pssm_matrix, ddgun_matrix), axis=1)
    # After concatenation, column 0 of second matrix is for distance, columns [1-6] are for center->neighbor transformation,
    # columns [7-12] are for neighbor->center transformation,
    # and columns [13-17] are for the 5 center-neighbor hydrogen bond features explained in the
    # ```calc_hbond_features_for_neighbor_list``` function defined above
    return first_matrix, second_matrix, gnn_edge_dict, center_icode


# Before executing this function, the PSSM files with naming format "pdbIdchain.pssm" needs to be stored
# in the ```pssm_dir``` which is a global variable
def returnPSSMArray(pdbIdPlusChain, pssm_dir, convert_upper=False):
    if convert_upper:
        fileName = pdbIdPlusChain.upper() + ".pssm"
    else:
        fileName = pdbIdPlusChain + ".pssm"
    try:
        fullPath = os.path.join(pssm_dir, fileName)
        f = open(fullPath)
    except:
        fileName = pdbIdPlusChain[0:4] + str(convertChainFromAlphabetToNumber(pdbIdPlusChain[4])) + ".pssm"
        fullPath = os.path.join(pssm_dir, fileName)
        f = open(fullPath)

    # #     all the target lines in the PSSM files have (2+20+20+2=44) strings after line.split()
    target_lines = [line.split() for line in f.readlines() if (len(line.split())) == 44]
    number_of_residues = len(target_lines)

    pssm_features = np.zeros((number_of_residues, 20))

    for idx, line in enumerate(target_lines):
        pssm_features[idx, :] = line[2:22]

    f.close()

    return pssm_features

# This function will return a vector of size 20 , where the position corresponding to ```reference``` has +1,
# position corresponding to ```alternate``` has -1, and all other positions have 0
# This function will use ```returnPSSMMapping``` function defined above for retrieving index from amino acid one letter code
def getFlagVector(reference, alternate):
    # Assuming, reference is one letter amino acid code
    reference_index = returnPSSMMapping(reference)
    alternate_index = returnPSSMMapping(alternate)
    flag_vec = np.zeros(shape=20,dtype=np.float32)
    flag_vec[reference_index] = 1
    flag_vec[alternate_index] = -1
    return flag_vec


# forward is a boolean parameter depicting state of the mutant (forward or reverse)
# mutant and wild are one letter amino acid codes
# This function returns a vector representing the mutation
def returnToFromFeatures(wild, mutant, forward):
    # Assuming that, prop_dict has three letter names of residues as keys
    mut_prop = np.array(getProps(mutant))
    wild_prop = np.array(getProps(wild))
    if forward:
        return wild_prop - mut_prop
    else:
        return mut_prop - wild_prop

# This function will fetch icodes for mutants where only amino acid and position info is available
def findICode(structure_dict,pdbIdPlusChain,amino_acid,position):
    structure = structure_dict[str(pdbIdPlusChain)]
    model = structure[0]
    # Doing this because of the single chain assumption
    chain = list(model.get_chains())[0]
    icode = " "

    for residue in chain:
        if residue.get_id()[1] == position and threeToOne(residue.get_resname()) == amino_acid:
            icode = residue.get_id()[2] if residue.get_id()[2] != " " else "Z"

    return icode

# This function will utilize functions above, create all the tensors required for training the model, and return those tensors

def return_all_training_related_tensors_for_S_669_dataset(pose_dict, structure_dict, two_level_dict, pssm_dir, distance, ddgun_dict,
                                                          add_rev=True, train=True):

    mutChains = []

        pdbIds.append(pdbIdChain[0:-1])
        mutChains.append(pdbIdChain[-1])

    mutations = list(dataset["PDB_Mut"])
    ddgs = list(dataset["DDG_checked_dir"])

    X_feature = []
    X_edge = []
    X_self = []
    X_prop_to_from = []
    y = []

    X_pssm = []
    X_alt = []
    X_flag = []
    X_edge_index_induced = []
    X_edge_index_complete = []
    X_edge_features_induced = []
    X_edge_features_complete = []

    if train:
        ddgun_ssl = []

        # ddg_avail, ssl_avail will be used for controlling the down and ssl loss functions
        ddg_avail = []
        ssl_avail = []

    for pdbId, mutChain, mutation, ddg in tqdm(zip(pdbIds, mutChains, mutations, ddgs)):
        pos = [int(s) for s in re.findall('-?\d+', mutation)][0]
        wild = mutation[0]
        mut = mutation[len(mutation) - 1]

        pdbId = pdbId.lower()

        # If some pdb file is not in the target directory for some reason, skip it
        if not os.path.isfile(os.path.join(pdbDirectory, pdbId + mutChain + ".pdb")):
            continue

        icode = findICode(structure_dict=structure_dict, pdbIdPlusChain=pdbId + mutChain, amino_acid=wild, position=pos)

        # Assuming the default icode here
        res = getNeighborMatrices(two_level_dict=two_level_dict, pose_dict=pose_dict,
                                                          structure_dict=structure_dict, ddgun_dict=ddgun_dict,
                                                          pdbIdPlusChain=pdbId + mutChain, position=pos, icode=icode,
                                  amino_acid=wild, pssm_dir=pssm_dir, distance=distance)

        global residue_missing

        # If the wildtype residue has been removed from structure, skip that mutation
        if residue_missing:
            continue

        feature_matrix, edge_matrix, gnn_edge_dict, icode = [], [], [], "Z"
        global no_neighbors
        # Skipping the no_neighbor cases for now
        if no_neighbors:
            continue
        else:
            feature_matrix, edge_matrix, gnn_edge_dict, icode = res

        # Not considering hyrodgen bond edge features since they seem to destabilize training
        edge_matrix = edge_matrix[:, 0:-5]

        edge_index_induced = gnn_edge_dict["edge_index_induced"]
        edge_index_complete = gnn_edge_dict["edge_index_complete"]
        edge_features_induced = gnn_edge_dict["edge_features_induced"]
        edge_features_complete = gnn_edge_dict["edge_features_complete"]
        # GNN edge information end

        if train:
            ddgun_pred_protchain = ddgun_dict.get(pdbId+mutChain,None)

        X_feature.append(feature_matrix)
        X_edge.append(edge_matrix)
        X_self.append(getProps(amino_acid=wild).reshape(1, -1))
        X_alt.append(getProps(amino_acid=mut).reshape(1, -1))
        X_flag.append(getFlagVector(wild, mut).reshape(1, -1))
        X_edge_index_induced.append(edge_index_induced)
        X_edge_index_complete.append(edge_index_complete)
        X_edge_features_induced.append(edge_features_induced)
        X_edge_features_complete.append(edge_features_complete)

        if train:
            ddg_avail.append(1)
            if ddgun_pred_protchain is not None:
                if icode !="Z":
                    ddgun_ssl.append(float(ddgun_pred_protchain[wild+str(pos)+icode+mut]))
                    ssl_avail.append(1)
                else:
                    ddgun_ssl.append(float(ddgun_pred_protchain[wild+str(pos)+mut]))
                    ssl_avail.append(1)
            else:
                # These cases will not be considered during training
                ddgun_ssl.append(0)
                ssl_avail.append(0)
        # The follwing lines are for the reverse mutation instances
        if add_rev:
            X_feature.append(feature_matrix)
            X_edge.append(edge_matrix)
            X_self.append(getProps(amino_acid=mut).reshape(1, -1))
            X_alt.append(getProps(amino_acid=wild).reshape(1, -1))
            X_flag.append(getFlagVector(mut, wild).reshape(1, -1))
            X_edge_index_induced.append(edge_index_induced)
            X_edge_index_complete.append(edge_index_complete)
            X_edge_features_induced.append(edge_features_induced)
            X_edge_features_complete.append(edge_features_complete)
            if train:
                ddg_avail.append(1)

        X_prop_to_from.append(returnToFromFeatures(wild=wild, mutant=mut, forward=True))
        if add_rev:
            X_prop_to_from.append(returnToFromFeatures(wild=wild, mutant=mut, forward=False))

        pssm_array = returnPSSMArray(pdbId + mutChain, pssm_dir=pssm_dir)

        position_pssm = pssm_array[(two_level_dict[pdbId + mutChain])[wild + str(pos)+icode]]
        wild_pssm = position_pssm[returnPSSMMapping(wild)]
        mut_pssm = position_pssm[returnPSSMMapping(mut)]

        X_pssm.append(np.array([wild_pssm, mut_pssm]))
        if add_rev:
            X_pssm.append(np.array([mut_pssm, wild_pssm]))

        y.append(ddg)
        if add_rev:
            y.append(ddg * (-1))

        pdbIds_avail.append(pdbId)
        if add_rev:
            pdbIds_avail.append(pdbId)

    t2 = time.time()
    print("Total time taken: {} minutes".format(round(((t2 - t1) / 60), 3)))

    if train:
        return X_feature, X_edge, X_self, X_prop_to_from, y, pdbIds_avail, X_pssm, X_alt, X_flag, \

def make_mutations(pdbId,ssl_directory,aa_percentage,site_percentage):
    mut_map = get_mut_map()

    parser = PDBParser(QUIET=True)
    structure = parser.get_structure(id=pdbId.split(".")[0],file=os.path.join(ssl_directory,pdbId))
    model = structure[0]
    residue_list = []

    # Assuming that the model contains only one chain
    for chain in model:
        for residue in chain:
            residue_list.append(residue)

    residue_list = random.sample(residue_list,int(len(residue_list)*(site_percentage/100)))

    mutations = []

    for residue in residue_list:
        one_letter_wild_code = three_to_one(residue.get_resname())
        seqres = residue.get_id()[1]
        icode = residue.get_id()[2] if residue.get_id()[2] != " " else "Z"
        local_mut_list = mut_map[one_letter_wild_code]
        local_mut_list = random.sample(local_mut_list,int(len(local_mut_list)*(aa_percentage/100)))
        for alternate in local_mut_list:
            mutations.append(one_letter_wild_code+str(seqres)+str(icode)+alternate)

    return mutations

def collate_fn_padd(batch):
    '''
    Padds batch of variable length

    note: it converts things ToTensor manually here since the ToTensor transform
    assume it takes in images rather than arbitrary tensors.
    '''
    node_features = [t[0] for t in batch]
    edge_features = [t[1] for t in batch]
    labels = torch.tensor([t[2] for t in batch])
    self_features = torch.tensor([t[3] for t in batch])
    to_from_features = torch.tensor([t[4] for t in batch])
    pssm_features = torch.tensor([t[5] for t in batch])
    alt_features = torch.tensor([t[6] for t in batch])
    flag_features = torch.tensor([t[7] for t in batch])
    # Check the length of the first tuple in the ```batch``` list to determine whether we are in train mode or not, and deal with ssl tensors accordingly
    # Assuming, the tuples in batch will have 11 elements during training, and 8 elements during testing after removing the ssl related tensors
    train_flag = True if (len(batch[0]) == 11) else False
    if train_flag:
        ddgun_ssl = torch.tensor([t[8] for t in batch])
        ddg_avail = torch.tensor([t[9] for t in batch])
        ssl_avail = torch.tensor([t[10] for t in batch])
        # edge_index_induced = torch.tensor([t[8] for t in batch],dtype=torch.long)
    # edge_index_complete = torch.tensor([t[9] for t in batch],dtype=torch.long)
    # edge_features_induced = torch.tensor([t[10] for t in batch])
    # edge_features_complete = torch.tensor([t[11] for t in batch])
    ## get number of neighbors in each of the instances
    lengths = torch.LongTensor([t.shape[0] for t in node_features])
    node_features = [torch.tensor(n) for n in node_features]
    edge_features = [torch.tensor(e) for e in edge_features]
    ## padd
    node_features = torch.nn.utils.rnn.pad_sequence(node_features, batch_first=True)
    edge_features = torch.nn.utils.rnn.pad_sequence(edge_features, batch_first=True)

    # return node_features, edge_features, labels, lengths, self_features, to_from_features, pssm_features, alt_features, flag_features, edge_index_induced, edge_index_complete, edge_features_induced, edge_features_complete
    if train_flag:
        return node_features, edge_features, labels, lengths, self_features, to_from_features, pssm_features, alt_features, flag_features, ddgun_ssl, ddg_avail, ssl_avail
    else:

# This function will return a two-level dictionary
# The outer level will have (pdbId+chain) as key, and inner-level will have (one letter wildtype code+seqres+ one letter alternate residue code) as key

# This function will read in all files from ddgun_directory 
def return_ddgun_map(ddgun_directory):
    two_level_dict = {}
    fnames = [n.split(".")[0] for n in os.listdir(ddgun_directory)]

    for name in fnames:

        f = open(os.path.join(ddgun_directory,name+".pdb.outs"),"r")
        dat = pd.read_csv(f,sep="\t")
        f.close()

        variants = dat["VARIANT"]
        ddgs = dat["S_DDG"]

        inner_dict = {}

        # Iterating through specific positions
        for vs,ds in zip(variants,ddgs):
            # Iterating through all mutations in that specific position
            for v,d in zip(vs.split(","),ds.split(",")):
                inner_dict[v] = float(d) * (-1)
        two_level_dict[name] = copy.deepcopy(inner_dict)
    return two_level_dict

# This will be used a submodel inside the regression/classification model
# This will process the input and create the environment representation using simple old school attention
# The outputs of this model will be concatenated with to_from_features for now inside the main regressio model
class Model_feat(nn.Module):
    # The default argument values are global variables declared in a cell above
    def __init__(self, n_node_props, n_edge_props, n_self_props, node_embedding_dimension, self_embedding_dimension,
                    edge_embedding_dimension, node_embedding_dropout=0):
        super(Model_feat, self).__init__()
        self.gat_embedding_dimension = 20
        self.node_embedding_dimension = node_embedding_dimension
        self.node_embedding_dropout = node_embedding_dropout
        # (+20) because the flag_vector has 20 entries
        self.node_embedding_transformation_2 = nn.Linear(in_features=self.gat_embedding_dimension+20+edge_embedding_dimension,
                                                            out_features=node_embedding_dimension)
    
        self.edge_embedding_transformation = nn.Linear(in_features=n_edge_props,
                                                        out_features=edge_embedding_dimension)

        self.gat_one = GATConv(in_channels=n_node_props,out_channels=self.gat_embedding_dimension,heads=2,concat=False,add_self_loops=False,edge_dim=n_edge_props)
        # self.gat_one = GATv2Conv(in_channels=n_node_props,out_channels=self.gat_embedding_dimension,heads=2,concat=False,add_self_loops=False)

    # n_neighbors is a 1D torch tensor of shape (batch_size,) which depicts number of neighbors for each of the instances
    # in the batch
    # feature_matrix has shape (max_n_neighbors,#node_embedding) or (max_n_neighbors,#edge_embedding)
    # will return a mask tensor having same shape as feature_matrix where activations coming from the padded neighbors are zeroed out
    def getMask(self, feature_matrix, n_neighbors):
        mask_tensor = []
        max_neighbors = feature_matrix.size()[1]
        n_props = feature_matrix.size()[2]
        # Creating the mask for one instance at a time
        for n in range(len(n_neighbors)):
            # print(type(n_neighbors[n]),n_neighbors[n].shape)
            cur_num = n_neighbors[n].item()
            # Here, assuming that "device" variable is available in the global scope
            # It might have to be sent through arguments later
            one_mask = torch.ones((cur_num, n_props), dtype=torch.float32, device=device)
            zero_mask = torch.zeros((max_neighbors - cur_num, n_props), dtype=torch.float32, device=device)
            mask_tensor.append(torch.cat([one_mask, zero_mask], dim=0))
        return torch.stack(mask_tensor)

    def convert_node_embedding_format(self,b_info,embedding_mat,batch_size):
        format_list = []
        for i in range(batch_size):
            format_list.append([])
        for index,g_num in enumerate(b_info):
            format_list[g_num].append(embedding_mat[index])
        for i in range(batch_size):
            format_list[i] = torch.stack(format_list[i])
        return torch.nn.utils.rnn.pad_sequence(format_list, batch_first=True)


    def forward(self, node_features, edge_features, lengths, self_features, alt_features, flag_features, data_induced, data_complete):
        n_neighbors = node_features.size()[1]

        edge_index_induced = data_induced.edge_index
        edge_features_induced = data_induced.edge_attr
        edge_index_complete = data_complete.edge_index
        edge_features_complete = data_complete.edge_attr
        node_features_induced = data_induced.x
        node_features_complete = data_complete.x
        b_info_induced = data_induced.batch
        b_info_complete = data_complete.batch

        max_allowed = node_features.size()[0] - 1
        max_referenced = edge_index_complete.max()

        # print("C1 : {}".format(node_features.dtype))
        node_features = self.gat_one(x=node_features_induced,edge_index=edge_index_induced,edge_attr=edge_features_induced)
        # print("C2 : {}".format(node_features.dtype))
        node_features = self.convert_node_embedding_format(b_info=b_info_induced,embedding_mat=node_features,batch_size=8)
        # print("C3 : {}".format(node_features.dtype))

        flag_features_repeat = flag_features.repeat(1, n_neighbors, 1)

        edge_transformed = self.edge_embedding_transformation(edge_features)
        edge_transformed = F.relu(edge_transformed)
        edge_transformed_masked = self.getMask(edge_transformed, lengths) * edge_transformed

        node_transformed = self.node_embedding_transformation_2(
            torch.cat([node_features, flag_features_repeat, edge_transformed_masked],
                        dim=-1))
        node_transformed = F.relu(node_transformed)
        node_transformed = F.dropout(node_transformed, p=self.node_embedding_dropout, training=self.training)
        node_transformed_masked = self.getMask(node_transformed, lengths) * node_transformed

        output_node_embedding = node_transformed_masked.sum(dim=-2)

        return output_node_embedding


class Model_regressor(nn.Module):
    # The default argument values are global variables declared in a cell above
    def __init__(self, n_node_props, n_edge_props, n_self_props, node_embedding_dimension,
                    edge_embedding_dimension, self_embedding_dimension,
                    dropout_rate=0.5, node_embedding_dropout=0):
        super(Model_regressor, self).__init__()
        self.dropout_rate = dropout_rate
        self.node_embedding_dimension = node_embedding_dimension
        self.model_feat = Model_feat(n_node_props=n_node_props, n_edge_props=n_edge_props,
                                        n_self_props=n_self_props,
                                        node_embedding_dimension=node_embedding_dimension,
                                        self_embedding_dimension= self_embedding_dimension,
                                        edge_embedding_dimension=edge_embedding_dimension,
                                        node_embedding_dropout=node_embedding_dropout)
        # (+2) for the two pssm values
        # Removing the use of to_from_features for now
        # self.fc1 = nn.Linear(in_features=node_embedding_dimension + 2, out_features=50)
        self.fc1_ddgun_pred = nn.Linear(in_features=node_embedding_dimension, out_features=50)
        self.fc1_ddg_pred = nn.Linear(in_features=node_embedding_dimension+2, out_features=50)

        self.fc2_ddg_pred = nn.Linear(in_features=50, out_features=1)
        # num_features 2 because of only 2 pssm values
        self.pssm_input_normalizer = nn.BatchNorm1d(num_features=2, affine=False)
        # adding edge normalization because hydrogen bond count features could be quite large(I think)
        self.edge_feature_normalizer = nn.BatchNorm1d(num_features=number_of_edge_features, affine=False)

        self.fc2_ddgun_pred = nn.Linear(in_features=50, out_features=1)

    # edge_features have shape (batch,#neighbors,#edge_features), node_features have shape (batch,#neighbors,#props)
    def forward(self, node_features, edge_features, lengths, self_features, to_from_features, pssm_features, alt_features, flag_features, data_induced, data_complete):
        # putting normalizer only on pssm now because it seems to have large scales
        # need to add normalizer to edge feature also, but cannot feagure out how pytorch BatchNorm1d exactly works (num_features is confusing me)
        # this link (https://discuss.pytorch.org/t/batchnorm1d-for-evaluation/67376/2) seems useful
        # need to check by separately running a sample of size (b,#neighbors,#edge_feature through BatchNorm1d)
        # this batchnorm did not help performance
        pssm_features = self.pssm_input_normalizer(pssm_features)
        # putting the permutation because ``BatchNorm1d`` expects the input to be of shape (N,C,L), but my ``edge_features`` shape is (N,L,C)
        edge_features = (self.edge_feature_normalizer(edge_features.permute(0, 2, 1).contiguous())).permute(0,
                                                                                                            2,
                                                                                                            1).contiguous()
        node_features = self.model_feat(node_features, edge_features, lengths, self_features, alt_features, flag_features, data_induced, data_complete)
        # env_to_from = 

        f = self.fc1_ddg_pred(torch.cat([node_features, pssm_features], dim=-1))
        f = F.relu(f)
        f_1_ddg_pred = F.dropout(f, p=self.dropout_rate, training=self.training)

        f = self.fc1_ddgun_pred(node_features)
        f = F.relu(f)
        f_1_ddgun_pred = F.dropout(f, p=self.dropout_rate, training=self.training)
        
        
        ddg_pred = self.fc2_ddg_pred(f_1_ddg_pred)

        ddgun_pred = self.fc2_ddgun_pred(f_1_ddgun_pred)
        return ddg_pred, ddgun_pred

def plot_grad_flow(named_parameters):
    '''Plots the gradients flowing through different layers in the net during training.
    Can be used for checking for possible gradient vanishing / exploding problems.
    
    Usage: Plug this function in Trainer class after loss.backwards() as 
    "plot_grad_flow(self.model.named_parameters())" to visualize the gradient flow'''
    ave_grads = []
    max_grads= []
    layers = []
    for n, p in named_parameters:
        if(p.requires_grad) and ("bias" not in n):
            layers.append(n)
            ave_grads.append(p.grad.abs().mean().cpu())
            max_grads.append(p.grad.abs().max().cpu())
    plt.bar(np.arange(len(max_grads)), max_grads, alpha=0.1, lw=1, color="c")
    plt.bar(np.arange(len(max_grads)), ave_grads, alpha=0.1, lw=1, color="b")
    plt.hlines(0, 0, len(ave_grads)+1, lw=2, color="k")
    plt.xticks(range(0,len(ave_grads), 1), layers, rotation="vertical")
    plt.xlim(left=0, right=len(ave_grads))
    plt.ylim(bottom = -0.001, top=0.08) # zoom in on the lower gradient regions
    plt.xlabel("Layers")
    plt.ylabel("average gradient")
    plt.title("Gradient flow")
    plt.grid(True)
    plt.legend([Line2D([0], [0], color="c", lw=4),
                Line2D([0], [0], color="b", lw=4),
                Line2D([0], [0], color="k", lw=4)], ['max-gradient', 'mean-gradient', 'zero-gradient'])
    plt.show()

def train(model, optimizer, dataLoader_one, dataLoader_two, dataLoader_three):
    model.train()
    for data_org, data_induced, data_complete in zip(dataLoader_one,dataLoader_two,dataLoader_three):
        node_features, edge_features, labels, lengths, self_features, to_from_features, pssm_features, alt_features, flag_features, ddgun_ssl, ddg_avail, ssl_avail = data_org
        with torch.no_grad():
            node_features, edge_features, labels, lengths, self_features, to_from_features, pssm_features, alt_features, flag_features, ddgun_ssl, ddg_avail, ssl_avail, \
            data_induced, data_complete  = \
            node_features.float().to(device), edge_features.float().to(device), labels.float().to(device), lengths.to(device), \
            self_features.float().to(device), to_from_features.float().to(device), pssm_features.float().to(device), alt_features.float().to(device), flag_features.float().to(device), \
            ddgun_ssl.float().to(device),ddg_avail.float().to(device),ssl_avail.float().to(device),data_induced.to(device), data_complete.to(device)    
            
            labels = labels.reshape(-1, 1)
            ddgun_ssl = ddgun_ssl.reshape(-1, 1)
        
        ddg_pred, ddgun_pred = model(node_features, edge_features, lengths, self_features, to_from_features, pssm_features, alt_features, flag_features, data_induced, data_complete)
        loss_ssl = F.mse_loss(ddgun_pred[ssl_avail==1], ddgun_ssl[ssl_avail==1], reduction="mean")
        loss_down = F.mse_loss(ddg_pred[ddg_avail==1], labels[ddg_avail==1], reduction="mean")
        loss = loss_ssl + loss_down
        optimizer.zero_grad()
        loss.backward()
        # plot_grad_flow(model.named_parameters())
        optimizer.step()

@torch.no_grad()
def eval(model, dataLoader_one, dataLoader_two, dataLoader_three):
    model.eval()
    total_loss_ssl = 0
    total_loss_down = 0
    total_exp_ddgs = 0
    total_ssl_ddgs = 0

    for data_org, data_induced, data_complete in zip(dataLoader_one,dataLoader_two,dataLoader_three):
        node_features, edge_features, labels, lengths, self_features, to_from_features, pssm_features, alt_features, flag_features, ddgun_ssl, ddg_avail, ssl_avail = data_org
        # Length should not be converted to float since it is used in size arguments
        node_features, edge_features, labels, lengths, self_features, to_from_features, pssm_features, alt_features, flag_features, ddg_avail, ddgun_ssl, \
        data_induced, data_complete  = \
        node_features.float().to(device), edge_features.float().to(device), labels.float().to(device), lengths.to(device), \
        self_features.float().to(device), to_from_features.float().to(device), pssm_features.float().to(device), alt_features.float().to(device), flag_features.float().to(device), \
        ddg_avail.float().to(device),ddgun_ssl.float().to(device),data_induced.to(device), data_complete.to(device)   
        
        labels = labels.reshape(-1, 1)
        ddgun_ssl = ddgun_ssl.reshape(-1, 1)
        ddg_pred, ddgun_pred = model(node_features, edge_features, lengths, self_features, to_from_features, pssm_features, alt_features, flag_features, data_induced, data_complete)
        loss_ssl = F.mse_loss(ddgun_pred[ssl_avail==1], ddgun_ssl[ssl_avail==1], reduction="mean")
        loss_down = F.mse_loss(ddg_pred[ddg_avail==1], labels[ddg_avail==1], reduction="mean")
        if (ssl_avail==1).sum().item() != 0:
            total_loss_ssl += loss_ssl * (ssl_avail==1).sum().item()
        if (ddg_avail==1).sum().item() != 0:
            total_loss_down += loss_down * (ddg_avail==1).sum().item()
        total_exp_ddgs = total_exp_ddgs + (ddg_avail==1).sum().item()
        total_ssl_ddgs = total_ssl_ddgs + (ssl_avail==1).sum().item()
    avg_loss_ssl = total_loss_ssl / total_ssl_ddgs
    avg_loss_down = total_loss_down / total_exp_ddgs
    return avg_loss_ssl.item(), avg_loss_down.item()


def train_epochs(model, dataLoader_train_one, dataLoader_train_two, dataLoader_train_three, dataLoader_val_one, dataLoader_val_two, dataLoader_val_three, train_args, log_dir, model_name):

    train_losses_ssl, train_losses_down, val_losses_ssl, val_losses_down  = [], [], [], []
    for epoch in range(epochs):

        train(model, optimizer, dataLoader_train_one, dataLoader_train_two, dataLoader_train_three)

            es(val_loss_down)

            if es.early_stop:
                break
            if es.counter == 0:
                cp(model)
