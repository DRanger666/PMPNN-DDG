class Person:
    def __init__(self,name,age):
        self.name = name
        self.age = age

    def __contains__(self,key):
        try:
            getattr(self,key)
            return True
        except:
            return False

    def __getitem__(self,key):
        print ("Inside `__getitem__` method!")
        if key in self:
            return getattr(self,key)
        else:
            return "DEFAULT IMAGINATION"
